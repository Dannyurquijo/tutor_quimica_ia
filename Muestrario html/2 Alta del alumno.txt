<script>
  let currentStep = 1;
  const totalSteps = 4;

  function updateStepUI() {
    // Hide all step panels
    for (let i = 1; i <= totalSteps; i++) {
      const panel = document.getElementById(`step-${i}`);
      if (panel) {
        if (i === currentStep) {
          panel.classList.remove('hidden');
          panel.classList.add('flex');
        } else {
          panel.classList.add('hidden');
          panel.classList.remove('flex');
        }
      }
    }

    // Update Top Step Navigation
    document.querySelectorAll('.step-nav').forEach((nav, idx) => {
      const stepNum = idx + 1;
      const numBadge = nav.querySelector('div:first-child');
      const tagSpan = nav.querySelector('span:first-child');

      if (stepNum === currentStep) {
        nav.classList.add('bg-surface-container-low');
        nav.classList.remove('opacity-75');
        if (numBadge) {
          numBadge.className = 'w-8 h-8 rounded-full bg-primary-container text-on-primary-container flex items-center justify-center font-label-md text-label-md font-bold shadow-sm';
        }
        if (tagSpan) {
          tagSpan.className = 'font-label-sm text-label-sm text-primary uppercase tracking-wider font-extrabold';
          tagSpan.innerText = `Paso ${stepNum} • Activo`;
        }
      } else if (stepNum < currentStep) {
        nav.classList.remove('bg-surface-container-low');
        nav.classList.remove('opacity-75');
        if (numBadge) {
          numBadge.className = 'w-8 h-8 rounded-full bg-tertiary-container text-on-tertiary-container flex items-center justify-center font-label-md text-label-md font-bold shadow-sm';
          numBadge.innerHTML = '<span class="material-symbols-outlined text-base">check</span>';
        }
        if (tagSpan) {
          tagSpan.className = 'font-label-sm text-label-sm text-tertiary uppercase tracking-wider font-bold';
          tagSpan.innerText = `Paso ${stepNum} • Listo`;
        }
      } else {
        nav.classList.remove('bg-surface-container-low');
        nav.classList.add('opacity-75');
        if (numBadge) {
          numBadge.className = 'w-8 h-8 rounded-full bg-surface-container-high text-on-surface-variant flex items-center justify-center font-label-md text-label-md font-bold';
          numBadge.innerText = stepNum;
        }
        if (tagSpan) {
          tagSpan.className = 'font-label-sm text-label-sm text-on-surface-variant/70 uppercase tracking-wider font-bold';
          tagSpan.innerText = `Paso ${stepNum}`;
        }
      }
    });

    // Handle Buttons
    const prevBtn = document.getElementById('prevBtn');
    const nextBtnText = document.getElementById('nextBtnText');
    const summaryName = document.getElementById('summaryName');
    const inputName = document.getElementById('inputName');

    if (inputName && summaryName) {
      summaryName.innerText = inputName.value || 'Sofía Morales';
    }

    if (currentStep === 1) {
      prevBtn.classList.add('hidden');
      nextBtnText.innerText = 'Continuar a Objetivos';
    } else {
      prevBtn.classList.remove('hidden');
      if (currentStep === 2) {
        nextBtnText.innerText = 'Continuar a Estilo & Avatar';
      } else if (currentStep === 3) {
        nextBtnText.innerText = 'Continuar a Calibración';
      } else if (currentStep === 4) {
        nextBtnText.innerText = 'Completar Registro y Comenzar Diagnóstico →';
      }
    }
  }

  function nextStep() {
    if (currentStep < totalSteps) {
      currentStep++;
      updateStepUI();
      window.scrollTo({ top: 0, behavior: 'smooth' });
